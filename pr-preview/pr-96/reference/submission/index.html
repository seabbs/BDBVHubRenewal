    Submission¶

    Write and check hub files and model metadata, and build the submitted round pages. See Getting started for a worked example.

    bdbvhubrenewal.submission ¶

    Submission: what goes to the hub and how it is checked.

    • hub: the hub file writer and the model metadata.
    • metadata: the metadata text of each model that may be submitted.
    • validate: the hub's own checks in Python (bdbvhubrenewal validate).
    • fitting: the fit configurations the submission CI runs.
    • rounds: the docs pages of submitted rounds.

    bdbvhubrenewal.submission.hub ¶

    Write hub submissions and model metadata.

    A submission is a CSV with columns reference_date, forecast_date, target, target_end_date, location, output_type, output_type_id and value. Every row is one week of one sample trajectory. forecast_date is the day the PR is opened. A resubmission in the same week keeps the file on the hub's main branch unchanged and appends rows with one new forecast_date. The round BDBV 2026 nowcasting and forecasting takes exactly 100 samples for each (reference_date, location, target), and each sample index covers the same target weeks within a location.

    submission.validate checks the written files as hubValidations does. The authoritative check is still hubValidations (task validate-hub).

    def zone_samples(
    name: str | None,
    task: ForecastTask,
    samples: pd.DataFrame,
    counts: pd.DataFrame,
    zones: tuple[str, ...],
    daily_zones: pd.DataFrame,

    """Return health-zone samples that sum to a model's province samples.

    A random walk model with no fit is run on the zones and summed to the
    provinces. Any other model's province samples, a fitted model's
    included, are split across the zones by `ZoneAllocator`, so no model
    is fitted here.

    Parameters
    ----------
    name : str or None
        The registered model, None for a fit configuration with none.
    task : ForecastTask
        The province task.
    samples : pandas.DataFrame
        The model's province samples for `task`.
    counts : pandas.DataFrame
        Weekly zone counts at the task's vintage (`weekly_zone_counts`).
    zones : tuple of str
        The hub's zone codes.
    daily_zones : pandas.DataFrame
        The daily zone table (`load_daily_zones`).

    Returns
    -------
    pandas.DataFrame
        Zone samples with `SAMPLE_COLUMNS`.
    """
    from bdbvhubrenewal.forecasters import get_model, list_models
    from bdbvhubrenewal.forecasters.randomwalk import RandomWalk
    from bdbvhubrenewal.forecasters.zones import ZoneAllocator, zone_forecast

    fitted = {s.name for s in list_models() if s.fitted}
    model = None if name is None else get_model(name)
    if isinstance(model, RandomWalk) and name not in fitted:
        return zone_forecast(model, task, samples, counts, zones, daily_zones)
    return ZoneAllocator().allocate(samples, task, counts, zones)

    def to_hub_table(
    samples: pd.DataFrame,
    reference_date: dt.date | str,
    target: str = "wk inc confirmed cases",
    *,
    unassigned: pd.DataFrame | None = None,
    zones: pd.DataFrame | None = None,
    n_samples: int = 100,
    forecast_date: dt.date | str | None = None,
) -> pd.DataFrame:
    """Turn forecast samples into the hub's submission table.

    The table meets every guarantee in `docs/submission/hub-spec.md`:
    sample ids `1..n_samples` reused in every location, one row per id for every
    week, the same weeks for every location, whole non-negative counts
    and the national value as the sum of the provinces for each sample.
    Any national rows in `samples` are replaced.

    Parameters
    ----------
    samples : pandas.DataFrame
        Columns `location`, `target_end_date`, `sample` (an integer
        index, one joint draw across weeks and provinces) and `value`.
    reference_date : datetime.date or str
        The round's reference date.
    target : str
        The target name.
    unassigned : pandas.DataFrame, optional
        A modelled component of the national count not assigned to any
        province, with columns `target_end_date`, `sample` and `value`.
        It is added to the national sum. Missing rows count as zero.
    zones : pandas.DataFrame, optional
        Health-zone samples (columns as `samples`), such as from
        `forecasters.zones.ZoneAllocator`. They must sum to the rounded
        province samples, sample by sample, and are not added to the
        national value.
    n_samples : int
        Number of samples each location must have.
    forecast_date : datetime.date or str, optional
        The day the PR is opened. The default is today (UTC), see
        `run_date`. The hub requires it to be on or before the
        reference date and within a day of the validation date.

    Returns
    -------
    pandas.DataFrame
        Columns `HUB_COLUMNS`, sorted by location, sample and week.
        Values are rounded to whole cases before the national sum, so the
        national value is exactly the sum of the written province values.

    Raises
    ------
    ValueError
        If a province or zone does not have sample ids `1..n_samples`, a
        sample lacks a week, locations cover different weeks, or the
        zones do not sum to their provinces.

    Examples
    --------
    >>> from bdbvhubrenewal.submission.hub import (
    ...     model_id, to_hub_table, write_submission
    ... )
    >>> samples = forecasts.query("model == 'randomwalknowcast'")
    >>> table = to_hub_table(samples, task.reference_date)
    >>> write_submission(table, "output/submission", model_id("rwnowcast"))
    """
    ref = as_date(reference_date).isoformat()
    fdate = as_date(forecast_date or run_date()).isoformat()
    prov = samples.loc[samples["location"] != NATIONAL].copy()
    if prov.empty:
        raise ValueError("No province samples to write.")
    prov["target_end_date"] = prov["target_end_date"].astype(str)
    prov["location"] = prov["location"].astype(str)
    prov["sample"] = prov["sample"].astype("int64")
    prov["value"] = prov["value"].clip(lower=0).round()
    zone = _zone_rows(prov, zones)
    ids = set(range(1, n_samples + 1))
    weeks = sorted(set(prov["target_end_date"]))
    for loc, rows in pd.concat([prov, zone]).groupby("location"):
        if set(rows["sample"]) != ids:
            raise ValueError(
                f"{loc}: sample ids must be 1 to {n_samples}, the same in "
                "every location."
            )
        if sorted(set(rows["target_end_date"])) != weeks:
            raise ValueError(f"{loc}: every location needs the same weeks.")
        if (
            len(rows) != n_samples * len(weeks)
            or rows.duplicated(["sample", "target_end_date"]).any()
        ):
            raise ValueError(
                f"{loc}: every sample needs exactly one row for each week."
            )
    national = (
        prov.groupby(["target_end_date", "sample"])["value"].sum().reset_index()
    )
    if unassigned is not None and not unassigned.empty:
        extra = (
            unassigned.assign(
                target_end_date=unassigned["target_end_date"].astype(str),
                sample=unassigned["sample"].astype("int64"),
                value=unassigned["value"].clip(lower=0).round(),
            )
            .groupby(["target_end_date", "sample"])["value"]
            .sum()
        )
        national = national.set_index(["target_end_date", "sample"])
        national["value"] = national["value"].add(extra, fill_value=0)
        national = national.reset_index()
    national["location"] = NATIONAL
    both = pd.concat([prov, national, zone], ignore_index=True)
    out = pd.DataFrame(
        {
            "reference_date": ref,
            "forecast_date": fdate,
            "target": target,
            "target_end_date": both["target_end_date"],
            "location": both["location"],
            "output_type": OUTPUT_TYPE,
            "output_type_id": both["sample"].astype("int64"),
            "value": both["value"].astype("int64"),
        }
    )
    return out.sort_values(
        ["location", "output_type_id", "target_end_date"]
    ).reset_index(drop=True)[list(HUB_COLUMNS)]

        The rounds in `hub-config/tasks.json`.
    admin : dict

    rounds: list[Round]

    @classmethod
    def load(cls, path: str | Path) -> Hub:
        """Read the hub configuration from a clone.

        Parameters
        ----------
        path : str or pathlib.Path
            Root of a clone of the hub repository.

        Returns
        -------
        Hub
            The configuration.

        Examples
        --------
        >>> from bdbvhubrenewal.submission.validate import Hub, validate_file
        >>> hub = Hub.load("../BDBV-Modeling-Hub")
        >>> report = validate_file("model-output/x/2026-11-07-x.csv", hub)
        >>> report.valid, [c.check for c in report.failed]
        """
        p = Path(path)
        tasks = json.loads((p / "hub-config" / "tasks.json").read_text())
        admin = json.loads((p / "hub-config" / "admin.json").read_text())
        schema_p = p / "hub-config" / "model-metadata-schema.json"
        schema = json.loads(schema_p.read_text()) if schema_p.exists() else None
        rounds = []
        for i, r in enumerate(tasks["rounds"]):
            mts = []
            for mt in r["model_tasks"]:
                tids = {
                    k: TaskId(
                        _vals(v.get("required")), _vals(v.get("optional"))
                    )
                    for k, v in mt["task_ids"].items()
                }
                ots = {
                    k: _parse_output_type(k, v)
                    for k, v in mt["output_type"].items()
                }
                mts.append(ModelTask(tids, ots))
            rounds.append(
                Round(
                    index=i,
                    name=r.get("round_name", f"round {i + 1}"),
                    round_id_from_variable=bool(r["round_id_from_variable"]),
                    round_id=r["round_id"],
                    model_tasks=mts,
                    submissions_due=r["submissions_due"],
                )
            )
        return cls(p, rounds, admin, schema)

        zoneinfo.ZoneInfo

    def submission_window(
        self, rnd: Round, round_id: str
    ) -> tuple[datetime, datetime]:
        """Open and close times of the submission window, in hub time.

        Matches hubValidations: the window opens at 00:00:00 on the start
        day and closes at 23:59:59 on the end day.

        Parameters
        ----------
        rnd : Round
            The round, from `find_round`.
        round_id : str
            Round id, the reference date.

        Returns
        -------
        tuple of datetime.datetime
            Opening and closing times.

    def validate_submission(
    file: str | Path,
    hub: Hub,
    metadata_dirs: Iterable[Path] = (),
    now: datetime | None = None,
    skip_window: bool = False,
    previous: str | Path | None = None,
) -> Report:
    """Run all model-output checks on one file.

    The hub's `forecast_date` checks run first. The standard checks then
    see only the new rows, without `forecast_date`, as in the hub's own
    workflow.

    Parameters
    ----------
    file : str or pathlib.Path
        The model-output CSV.
    hub : Hub
        The hub configuration.
    metadata_dirs : iterable of pathlib.Path
        Directories to look for the model's metadata in.
    now : datetime.datetime, optional
        Time to check the submission window against. Defaults to now.
    skip_window : bool, default False
        Skip the window check and the check that the new forecast date
        is recent.
    previous : str or pathlib.Path, optional
        The file now on the hub's main branch at the same path, if any.

    Returns
    -------
    Report
        The checks and their outcomes.
    """
    fp = Path(file)
    rep = Report(str(file))

    # File-level checks.
    meta = parse_file_name(fp.name)
    if not rep.add(
        "file_name",
        meta is not None,
        f'File name "{fp.name}" is valid.',
        halt=True,
        fail_message=f'File name "{fp.name}" must be '
        "<round_id>-<team>-<model>.<ext>.",
    ):
        return rep
    assert meta is not None
    model_id = f"{meta['team']}-{meta['model']}"
    round_id = meta["round_id"]
    rep.add(
        "file_location",
        fp.parent.name == model_id,
        "File directory name matches `model_id` in file name.",
        fail_message=f'File should be in directory "{model_id}", '
        f'not "{fp.parent.name}".',
    )
    rnd = hub.find_round(round_id)
    if not rep.add(
        "round_id_valid",
        rnd is not None,
        "`round_id` is valid.",
        halt=True,
        fail_message=f'`round_id` "{round_id}" is not in '
        "hub-config/tasks.json.",
    ):
        return rep
    assert rnd is not None
    if not rep.add(
        "file_format",
        meta["ext"] in hub.file_formats,
        "File is accepted hub format.",
        halt=True,
        fail_message=f'Extension "{meta["ext"]}" not in {hub.file_formats}.',
    ):
        return rep
    clock = now or datetime.now(hub.timezone)
    if clock.tzinfo is None:
        clock = clock.replace(tzinfo=hub.timezone)
    dirs = [*metadata_dirs, hub.path / "model-metadata"]
    rep.add(
        "metadata_exists",
        find_metadata(model_id, dirs) is not None,
        "Metadata file exists.",
        fail_message=f"No metadata file {model_id}.yml or .yaml.",
    )
    if skip_window:
        rep.skip("submission_time", "Submission window check skipped.")
    else:
        start, end = hub.submission_window(rnd, round_id)
        t = clock
        rep.add(
            "submission_time",
            start <= t <= end,
            "Submission time is within the submission window.",
            fail_message=f"Time {t.isoformat()} is outside window "
            f"{start.isoformat()} to {end.isoformat()}.",
        )

    # Read.
    try:
        with fp.open(newline="", encoding="utf-8-sig") as fh:
            reader = csv.DictReader(fh)
            cols = list(reader.fieldnames or [])
            rows: list[Row] = [dict(r) for r in reader]
    except (OSError, csv.Error, UnicodeDecodeError) as e:
        rep.add(
            "file_read",
            False,
            "",
            halt=True,
            fail_message=f"File could not be read: {e}",
        )
        return rep
    date_ids = _date_task_ids(rnd)
    bad_read = []
    for i, r in enumerate(rows, start=2):
        if None in r or any(v is None for v in r.values()):
            bad_read.append(f"line {i}: wrong number of fields")
            continue
        v = r.get("value", "")
        if v not in NA_STRINGS and not _is_number(v):
            bad_read.append(f'line {i}: `value` "{v}" is not numeric')
        for c in date_ids & set(cols):
            if r[c] not in NA_STRINGS and not DATE_RE.match(r[c]):
                bad_read.append(f'line {i}: `{c}` "{r[c]}" is not a date')
    if not rep.add(
        "file_read",
        not bad_read,
        "File could be read.",
        halt=True,
        fail_message="File could not be read with the hub schema: "
        + "; ".join(bad_read[:5]),
    ):
        return rep

    # The hub's own forecast_date checks, then the standard ones on the
    # rows this submission adds, without that column. Files of the
    # round's first, cumulative target have neither column and are not
    # subject to them.
    if FORECAST_DATE in cols or "target_end_date" in cols:
        prev_cols: list[str] | None = None
        prev_rows: list[Row] | None = None
        if previous is not None:
            try:
                prev_cols, prev_rows = read_rows(previous)
            except (OSError, csv.Error, UnicodeDecodeError) as e:
                rep.add(
                    "forecast_date_preserved",
                    False,
                    "",
                    halt=True,
                    fail_message=f"The hub's copy could not be read: {e}",
                )
                return rep
        today = None if skip_window else clock.astimezone(hub.timezone).date()
        problems, new_idx = forecast_date_problems(
            rows, cols, prev_rows, prev_cols, today
        )
        for check, message in problems:
            rep.add(check, False, "", halt=True, fail_message=message)
        if problems:
            return rep
        rep.add(
            "forecast_date",
            True,
            "`forecast_date` is valid and adds one new date.",
        )
        rows = [
            {k: v for k, v in rows[i].items() if k != FORECAST_DATE}
            for i in new_idx
        ]
        cols = [c for c in cols if c != FORECAST_DATE]

    # Round id column.
    ric = rnd.round_id
    if rnd.round_id_from_variable:
        if not rep.add(
            "valid_round_id_col",
            ric in cols,
            "`round_id_col` name is valid.",
            halt=True,
            fail_message=f'Column "{ric}" missing.',
        ):
            return rep
        ids = sorted({r[ric] for r in rows})
        if not rep.add(
            "unique_round_id",
            len(ids) == 1,
            f'`round_id` column "{ric}" has a single value.',
            halt=True,
            fail_message=f'Column "{ric}" contains {len(ids)} '
            f"values: {', '.join(ids)}.",
        ):
            return rep
        if not rep.add(
            "match_round_id",
            ids[0] == round_id,
            "`round_id` column matches the file name.",
            halt=True,
            fail_message=f"`round_id` value {ids[0]} does not "
            f'match file name "{round_id}".',
        ):
            return rep

    # Columns.
    expected = [*rnd.task_id_names, *STD_COLS]
    missing = [c for c in expected if c not in cols]
    extra = [c for c in cols if c not in expected]
    msg = []
    if missing:
        msg.append(f"Missing column(s): {', '.join(missing)}.")
    if extra:
        msg.append(f"Unexpected column(s): {', '.join(extra)}.")
    if not rep.add(
        "colnames",
        not msg,
        "Column names are valid.",
        halt=True,
        fail_message=" ".join(msg),
    ):
        return rep

    # Valid values and combinations.
    mt_idx = [_row_model_task(r, rnd) for r in rows]
    bad_rows = [r for r, m in zip(rows, mt_idx, strict=True) if m is None]
    if not rep.add(
        "valid_vals",
        not bad_rows,
        "`tbl` contains valid values/value combinations.",
        halt=True,
        fail_message=_invalid_values(
            bad_rows, rnd, [*rnd.task_id_names, "output_type"]
        ),
    ):
        return rep

    # Unique rows.
    key_cols = [*rnd.task_id_names, "output_type", "output_type_id"]
    keys = Counter(tuple(norm(r[c]) for c in key_cols) for r in rows)
    dups = sum(n - 1 for n in keys.values() if n > 1)
    rep.add(
        "rows_unique",
        dups == 0,
        "All task id/output type/output type id combinations are unique.",
        fail_message=f"{dups} duplicated row(s).",
    )

    _check_req_vals(rep, rows, rnd, mt_idx)
    _check_values(rep, rows, rnd, mt_idx)
    _check_non_desc(rep, rows, rnd)
    if any(r["output_type"] == "sample" for r in rows):
        _check_samples(rep, rows, rnd, mt_idx)
    else:
        for c in (
            "spl_mt_unique",
            "spl_compound_taskid_set",
            "spl_compound_tid",
            "spl_non_compound_tid",
            "spl_n",
        ):
            rep.skip(c, "No samples found.")
    _check_readme_horizon(rep, rows, round_id)
    return rep

            "sampler": dict(sampler),

    )

        fh.write(f"config={config.name} done\n")

    Docs pages for the submitted rounds, from the submission ledger.

    The submission workflow records every live round on the submission-ledger branch: state.json holds each round's outcome, and archive/<reference_date>/<run>/ holds the files of each run (the hub file, the plot, the provenance and the pull request body).

    Each round records every hub model it submitted under models, keyed by hub model id, with the folder of that model's archived files. make_round_pages turns that into one page per round at docs/submission/rounds/<reference_date>.md, served at /submission/rounds/<reference_date>/, which the hub pull requests link to, with one tab per hub model id.         without `state.json`, means no round has been submitted.
    docs : pathlib.Path or str
        The docs folder. Pages go to `submission/rounds/` inside it.
    generated : pathlib.Path or str
        Where the `latest-submission.md` snippet goes.
    pr_state : callable
        Returns a hub pull request's state from its URL.

    Returns
    -------
    list of str
        The reference dates with a page, newest first.
    """
    out = Path(docs) / "submission" / "rounds"
    if out.exists():
        shutil.rmtree(out)
    out.mkdir(parents=True)
    root = Path(ledger or ".")
    state = _read(root / "state.json") if ledger else {}
    rounds: dict[str, Any] = state.get("rounds", {})
    done = sorted(rounds, reverse=True)
    frozen = root / FROZEN
    for ref in done:
        if (frozen / f"{ref}.md").exists():
            for f in frozen.glob(f"{ref}*"):
                shutil.copy(f, out / f.name)
        else:
            _render(root, ref, rounds[ref], out)
    index = ["# Submission log", ""]
    if done:
        index += [
            "Every round each hub model was submitted for, newest first.",
            "The round links to its page.",
            "",
        ]
        index += _log(rounds, root, pr_state)
    (out / "index.md").write_text("\n".join(index) + "\n")
    snippet = Path(generated) / "latest-submission.md"
    snippet.parent.mkdir(parents=True, exist_ok=True)
    if done:
        ref = done[0]
        models = rounds[ref].get("models", {})
        each = ", ".join(
            f"{model_id} {m.get('status', '')} "
            f"({m.get('produced_by') or 'n/a'})"
            for model_id, m in sorted(models.items())
        )
        snippet.write_text(
            f"**Latest submission:** "
            f"[round {ref}](submission/rounds/{ref}.md): {each}. "
            "[Submission log](submission/rounds/index.md).\n"
        )
    else:
        snippet.write_text("")
    return done
